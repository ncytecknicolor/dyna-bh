# Apache para la versión de PRUEBA del sitio en dynamedia.mx/new (Bluehost)
Options -MultiViews
DirectorySlash Off
DirectoryIndex index.html

ErrorDocument 404 /new/404.html

RewriteEngine On
RewriteBase /new/

# Siempre HTTPS
RewriteCond %{HTTPS} off
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]

# Formulario de contacto
RewriteRule ^api/contact/?$ api/contact.php [L]

# Redirecciones
RewriteRule ^areas/desarrollo-web/?$ /new/areas/diseno-web [R=301,L]

# Páginas sin ".html" (/new/servicios -> servicios.html)
RewriteCond %{REQUEST_FILENAME}.html -f
RewriteRule ^(.+?)/?$ $1.html [L]

<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/jpeg "access plus 1 week"
  ExpiresByType image/png "access plus 1 week"
  ExpiresByType image/webp "access plus 1 week"
  ExpiresByType text/css "access plus 1 week"
  ExpiresByType application/javascript "access plus 1 week"
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>
